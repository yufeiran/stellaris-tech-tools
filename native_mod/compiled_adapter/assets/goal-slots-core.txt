country_event = {
 id = ttegoal.10 hide_window = yes is_triggered_only = yes
 immediate = {
  ttegs_recheck = yes
  if = {
   limit = { check_variable = { which = ttegs_count value = 1 } check_variable = { which = ttegs_destination value >= 1 } check_variable = { which = ttegs_destination value <= 5 } has_country_flag = ttegs_pending_valid }
   if = { limit = { check_variable = { which = ttegs_origin value = ttegs_destination } } ttegs_cancel = yes }
   else = {
    {{OCCUPIED}}
    if = { limit = { has_country_flag = ttegs_occupied }
     set_country_flag = ttegs_dialog
     country_event = { id = ttegoal.1 }
    }
    else = { ttegs_commit = yes }
   }
  }
  else = { ttegs_cancel = yes }
 }
}
